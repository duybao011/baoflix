"use client";

import Link from "next/link";
import { useEffect, useMemo, useState } from "react";
import FavoriteButton from "@/components/FavoriteButton";
import type { EpisodeServer, MovieDetail } from "@/lib/kkphim";
import { getImageUrl } from "@/lib/kkphim";
import { isTvModeActive } from "@/lib/tvMode";

type TvMovieDetailShellProps = {
  movie: MovieDetail;
  servers: EpisodeServer[];
  content: string;
};

type ServerPreview = {
  server: EpisodeServer;
  serverIndex: number;
  episodes: NonNullable<EpisodeServer["server_data"]>;
};

const TV_FOCUS_CLASS =
  "focus-visible:scale-[1.045] focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-yellow-300/80 focus-visible:ring-offset-2 focus-visible:ring-offset-black";

function getServerPreviews(servers: EpisodeServer[]): ServerPreview[] {
  return servers
    .map((server, serverIndex) => ({
      server,
      serverIndex,
      episodes: server.server_data ?? [],
    }))
    .filter((item) => item.episodes.length > 0);
}

function getEpisodeHref(movieSlug: string, serverIndex: number, episodeIndex: number) {
  return `/xem/${movieSlug}?server=${serverIndex}&tap=${episodeIndex}`;
}

function getFirstWatchHref(movieSlug: string, servers: EpisodeServer[]) {
  const firstServer = getServerPreviews(servers)[0];
  if (!firstServer) return "";
  return getEpisodeHref(movieSlug, firstServer.serverIndex, 0);
}

function getLatestEpisodeHref(movieSlug: string, servers: EpisodeServer[]) {
  let bestServerIndex = -1;
  let bestEpisodeIndex = -1;
  let bestEpisodeCount = 0;

  servers.forEach((server, serverIndex) => {
    const episodes = server.server_data ?? [];

    if (episodes.length > bestEpisodeCount) {
      bestServerIndex = serverIndex;
      bestEpisodeIndex = episodes.length - 1;
      bestEpisodeCount = episodes.length;
    }
  });

  if (bestServerIndex < 0 || bestEpisodeIndex < 0) return "";
  return getEpisodeHref(movieSlug, bestServerIndex, bestEpisodeIndex);
}

function normalizeServerName(name?: string) {
  const text = String(name || "").toLowerCase();

  if (text.includes("lồng") || text.includes("long")) return "Lồng tiếng";
  if (text.includes("thuyết") || text.includes("thuyet")) return "Thuyết minh";
  if (text.includes("vietsub") || text.includes("sub")) return "Vietsub";

  return name || "Server";
}

function InfoPill({ children, hot }: { children: React.ReactNode; hot?: boolean }) {
  return (
    <span
      className={[
        "rounded-full px-2.5 py-1 text-[10px] font-black",
        hot ? "bg-yellow-300 text-black" : "border border-white/10 bg-black/30 text-slate-200",
      ].join(" ")}
    >
      {children}
    </span>
  );
}

function ActionLink({
  href,
  children,
  tone = "soft",
  tvDefault,
}: {
  href: string;
  children: React.ReactNode;
  tone?: "primary" | "red" | "soft";
  tvDefault?: boolean;
}) {
  return (
    <Link
      href={href}
      prefetch={false}
      data-tv-default={tvDefault ? true : undefined}
      className={[
        "rounded-xl px-4 py-2.5 text-sm font-black transition duration-100",
        tone === "primary" ? "bg-yellow-300 text-black hover:bg-yellow-200" : "",
        tone === "red" ? "bg-red-600 text-white hover:bg-red-500" : "",
        tone === "soft" ? "border border-white/10 bg-white/5 text-white hover:bg-white/10" : "",
        TV_FOCUS_CLASS,
      ].join(" ")}
    >
      {children}
    </Link>
  );
}

function EpisodeButton({
  href,
  children,
  active,
}: {
  href: string;
  children: React.ReactNode;
  active?: boolean;
}) {
  return (
    <Link
      href={href}
      prefetch={false}
      className={[
        "flex min-h-[34px] items-center justify-center rounded-lg border px-1.5 text-center text-[10px] font-black transition duration-100 min-[1280px]:min-h-[36px] min-[1280px]:text-[11px]",
        active
          ? "border-yellow-300 bg-yellow-300 text-black"
          : "border-white/10 bg-white/5 text-white hover:border-yellow-300 hover:bg-white/10",
        TV_FOCUS_CLASS,
      ].join(" ")}
    >
      <span className="line-clamp-1">{children}</span>
    </Link>
  );
}

export default function TvMovieDetailShell({
  movie,
  servers,
  content,
}: TvMovieDetailShellProps) {
  const [enabled, setEnabled] = useState(false);
  const [activeServerIndex, setActiveServerIndex] = useState(0);

  useEffect(() => {
    function refresh() {
      setEnabled(isTvModeActive());
    }

    refresh();
    window.addEventListener("baoflix-tv-mode-change", refresh);
    window.addEventListener("storage", refresh);
    window.addEventListener("focus", refresh);

    return () => {
      window.removeEventListener("baoflix-tv-mode-change", refresh);
      window.removeEventListener("storage", refresh);
      window.removeEventListener("focus", refresh);
    };
  }, []);

  const firstWatchHref = useMemo(
    () => getFirstWatchHref(movie.slug, servers),
    [movie.slug, servers]
  );

  const latestEpisodeHref = useMemo(
    () => getLatestEpisodeHref(movie.slug, servers),
    [movie.slug, servers]
  );

  const serverPreviews = useMemo(() => getServerPreviews(servers), [servers]);
  const totalEpisodeCount = serverPreviews.reduce(
    (total, item) => total + item.episodes.length,
    0
  );

  useEffect(() => {
    if (activeServerIndex >= serverPreviews.length) {
      setActiveServerIndex(0);
    }
  }, [activeServerIndex, serverPreviews.length]);

  const activeServer = serverPreviews[activeServerIndex];

  if (!enabled) return null;

  return (
    <>
      <style>{`
        [data-baoflix-normal-detail='true'] {
          display: none !important;
        }
      `}</style>

      <section
        data-tv-scope="movie-detail-tv"
        data-tv-lock="true"
        data-tv-autofocus="true"
        className="baoflix-tv-page space-y-4"
      >
        <div className="grid gap-4 rounded-2xl border border-white/10 bg-gradient-to-br from-red-600/14 via-white/[0.03] to-yellow-300/10 p-3 lg:grid-cols-[150px_1fr] min-[1280px]:grid-cols-[170px_1fr] min-[1280px]:p-4">
          <div className="mx-auto w-[138px] overflow-hidden rounded-xl border border-white/10 bg-black/30 lg:w-full">
            <img
              src={getImageUrl(movie.poster_url || movie.thumb_url)}
              alt={movie.name}
              loading="lazy"
              decoding="async"
              className="aspect-[2/3] w-full object-cover"
            />
          </div>

          <div className="flex min-w-0 flex-col justify-center">
            <h1 className="line-clamp-2 text-2xl font-black leading-tight min-[1280px]:text-3xl">
              {movie.name}
            </h1>

            {movie.origin_name && (
              <p className="mt-1 line-clamp-1 text-sm font-bold text-yellow-300">
                {movie.origin_name}
              </p>
            )}

            <div data-tv-row className="mt-2 flex flex-wrap gap-1.5">
              {movie.year && <InfoPill>{movie.year}</InfoPill>}
              {movie.quality && <InfoPill>{movie.quality}</InfoPill>}
              {movie.lang && <InfoPill>{movie.lang}</InfoPill>}
              {movie.episode_current && <InfoPill hot>{movie.episode_current}</InfoPill>}
              {serverPreviews.length > 1 && <InfoPill hot>{serverPreviews.length} nguồn</InfoPill>}
            </div>

            <p className="mt-2 line-clamp-2 max-w-5xl text-sm leading-6 text-slate-300">
              {content || "Chưa có mô tả."}
            </p>

            <div data-tv-row data-tv-row-wrap="true" className="mt-3 flex flex-wrap gap-2">
              {firstWatchHref && (
                <ActionLink href={firstWatchHref} tone="primary" tvDefault>
                  ▶ Xem ngay
                </ActionLink>
              )}

              {latestEpisodeHref && latestEpisodeHref !== firstWatchHref && (
                <ActionLink href={latestEpisodeHref} tone="red">
                  Tập mới
                </ActionLink>
              )}

              <FavoriteButton movie={movie} />

              <ActionLink href="/tv">TV Hub</ActionLink>
            </div>
          </div>
        </div>

        {serverPreviews.length > 0 && (
          <section className="rounded-2xl border border-white/10 bg-white/[0.035] p-3 min-[1280px]:p-4">
            <div className="mb-3 flex items-end justify-between gap-3">
              <div>
                <h2 className="text-lg font-black min-[1280px]:text-xl">Chọn nguồn & tập</h2>
                <p className="mt-0.5 text-[11px] text-slate-400">
                  {serverPreviews.length} nguồn • {totalEpisodeCount} tập
                </p>
              </div>
            </div>

            <div data-tv-row data-tv-row-wrap="true" className="mb-3 flex flex-wrap gap-2">
              {serverPreviews.map((item, index) => (
                <button
                  key={`${item.server.server_name}-${item.serverIndex}`}
                  type="button"
                  onClick={() => setActiveServerIndex(index)}
                  data-tv-focus-key={`detail-source:${item.serverIndex}`}
                  className={[
                    "rounded-xl border px-3 py-2 text-xs font-black transition duration-100 min-[1280px]:text-sm",
                    index === activeServerIndex
                      ? "border-yellow-300 bg-yellow-300 text-black"
                      : "border-white/10 bg-white/5 text-white hover:bg-white/10",
                    TV_FOCUS_CLASS,
                  ].join(" ")}
                >
                  {normalizeServerName(item.server.server_name)}
                  <span className="ml-1 opacity-70">• {item.episodes.length}</span>
                </button>
              ))}
            </div>

            {activeServer && (
              <>
                <div data-tv-row className="mb-3 grid grid-cols-2 gap-2 sm:grid-cols-4">
                  <ActionLink
                    href={getEpisodeHref(movie.slug, activeServer.serverIndex, 0)}
                    tone="primary"
                  >
                    Tập đầu
                  </ActionLink>

                  <ActionLink
                    href={getEpisodeHref(
                      movie.slug,
                      activeServer.serverIndex,
                      activeServer.episodes.length - 1
                    )}
                    tone="red"
                  >
                    Tập mới
                  </ActionLink>

                  {activeServer.episodes[Math.floor(activeServer.episodes.length / 2)] && (
                    <ActionLink
                      href={getEpisodeHref(
                        movie.slug,
                        activeServer.serverIndex,
                        Math.floor(activeServer.episodes.length / 2)
                      )}
                    >
                      Giữa phim
                    </ActionLink>
                  )}

                  <ActionLink href="/tv">Thoát</ActionLink>
                </div>

                <div
                  data-tv-row
                  data-tv-row-wrap="true"
                  className="grid grid-cols-5 gap-1.5 sm:grid-cols-6 md:grid-cols-8 xl:grid-cols-10"
                >
                  {activeServer.episodes.slice(0, 80).map((episode, index) => (
                    <EpisodeButton
                      key={`${activeServer.serverIndex}-${episode.name}-${index}`}
                      href={getEpisodeHref(movie.slug, activeServer.serverIndex, index)}
                    >
                      {episode.name || `Tập ${index + 1}`}
                    </EpisodeButton>
                  ))}
                </div>

                {activeServer.episodes.length > 80 && (
                  <p className="mt-2 text-[11px] text-slate-500">
                    Đang hiển thị 80 tập đầu để TV mượt hơn. Dùng Tập mới để nhảy tập cuối.
                  </p>
                )}
              </>
            )}
          </section>
        )}

        <section className="rounded-2xl border border-white/10 bg-white/[0.025] p-3">
          <h2 className="text-base font-black">Thông tin nhanh</h2>

          <div className="mt-2 grid gap-2 text-xs text-slate-300 md:grid-cols-2">
            {movie.time && (
              <p>
                <span className="font-black text-white">Thời lượng:</span> {movie.time}
              </p>
            )}
            {movie.status && (
              <p>
                <span className="font-black text-white">Trạng thái:</span> {movie.status}
              </p>
            )}
            {movie.country && movie.country.length > 0 && (
              <p className="line-clamp-1">
                <span className="font-black text-white">Quốc gia:</span>{" "}
                {movie.country.map((item) => item.name).join(", ")}
              </p>
            )}
            {movie.category && movie.category.length > 0 && (
              <p className="line-clamp-1">
                <span className="font-black text-white">Thể loại:</span>{" "}
                {movie.category.map((item) => item.name).join(", ")}
              </p>
            )}
          </div>
        </section>
      </section>
    </>
  );
}
